=== Creating redis data
docker run --name my-redis -d -p 6379:6379 redis:8
docker run --rm --network container:my-redis redis redis-cli -h localhost SET users:4 '{"id":4, "name":"Nishiwaki", "email":"aoaoan@neo.net", "age":49}'
docker run --rm --network container:my-redis redis redis-cli -h localhost SET users:1 '{"id":1, "name":"Alice", "email":"alice@example.com", "age":30}'

docker run --rm --network container:dbcache_redis redis redis-cli -h localhost GET users:1

docker exec -it my-redis redis-cli BGSAVE
docker exec -it my-redis redis-cli CONFIG GET dir
docker cp my-redis:/data/dump.rdb .

=== Creating postgres data
docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "CREATE TABLE users (id SERIAL PRIMARY KEY, name TEXT, email TEXT, age INT);"

docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "INSERT INTO users (name, email, age) VALUES ('Alice', 'alice@example.com', 30);"

docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "INSERT INTO users (name, email, age) VALUES ('Oomoto', 'nocchi@perfume.com', 36);"

docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "INSERT INTO users (name, email, age) VALUES ('Kashino', 'kashiyuka@europe.gov', 22);"

docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "INSERT INTO users (name, email, age) VALUES ('Nishiwaki', 'aoaoan@neo.net', 49);"

docker exec -t dbcache_postgres pg_dump -U local_dev -d local_db > dump.sql
