docker run --rm --network container:my-redis redis redis-cli -h localhost SET users:4 '{"id":4, "name":"Nishiwaki", "email":"aoaoan@neo.net", "age":49}'

=== Creating postgres data
docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "CREATE TABLE users (id SERIAL PRIMARY KEY, name TEXT, email TEXT, age INT);"

docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "INSERT INTO users (name, email, age) VALUES ('Alice', 'alice@example.com', 30);"

docker run --rm --network container:dbcache_postgres postgres psql -h localhost -U local_dev -d local_db -c "INSERT INTO users (name, email, age) VALUES ('Oomoto', 'nocchi@perfume.com', 36);"
